cmake_minimum_required(VERSION 3.20)
project(aeroloop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(frame_tests tests/cpp/frame_tests.cpp)
target_include_directories(frame_tests PRIVATE firmware/control_core)
add_test(NAME frame_contract COMMAND frame_tests)
add_library(aeroloop_rate SHARED firmware/control_core/rate_api.cpp)
target_include_directories(aeroloop_rate PRIVATE firmware/control_core)
add_executable(rate_tests tests/cpp/rate_tests.cpp)
target_include_directories(rate_tests PRIVATE firmware/control_core)
add_test(NAME rate_controller COMMAND rate_tests)
if(MSVC)
  target_compile_options(aeroloop_rate PRIVATE /W4 /WX)
  target_compile_options(rate_tests PRIVATE /W4 /WX)
else()
  target_compile_options(aeroloop_rate PRIVATE -Wall -Wextra -Werror)
  target_compile_options(rate_tests PRIVATE -Wall -Wextra -Werror)
endif()
